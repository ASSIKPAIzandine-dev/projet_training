
// ---- Écriture dans un fichier ----

// Ouvrir le fichier en mode écriture ('w')
// 'w' : ouvre pour écriture. Le pointeur est au début du fichier. S'il existe, le contenu est effacé. S'il n'existe pas, il est créé.
$fp_ecriture = fopen('mon_fichier_avance.txt', 'w');

if ($fp_ecriture) {
    // Écrire des données dans le fichier
    fwrite($fp_ecriture, "Première ligne écrite.\n");
    fwrite($fp_ecriture, "Deuxième ligne écrite.\n");

    // Fermer le fichier
    fclose($fp_ecriture);
    echo "Données écrites dans mon_fichier_avance.txt<br>";
} else {
    echo "Erreur : Impossible d'ouvrir le fichier pour écrire.<br>";
}

// ---- Lecture d'un fichier ligne par ligne ----
